        "import gradio as gr\n",
        "def calculate(expression):\n",
        "    try:\n",
        "        # Only allow safe characters\n",
        "        allowed = \"0123456789+-*/.() %\"\n",
        "        if not all(c in allowed for c in expression):\n",
        "            return \"Error\"\n",
        "        # Replace % with /100 for percentage-style calculations\n",
        "        expr = expression.replace(\"%\", \"/100\")\n",
        "        result = eval(expr)\n",
        "        # Format with commas, like the screenshot (3,230,000)\n",
        "        if isinstance(result, float) and result.is_integer():\n",
        "            result = int(result)\n",
        "        return f\"{result:,}\"\n",
        "    except Exception:\n",
        "        return \"Error\"\n",
        "class CalcState:\n",
        "    def __init__(self):\n",
        "        self.expr = \"\"\n",
        "state = CalcState()\n",
        "def press(btn):\n",
        "    if btn == \"C\":\n",
        "        state.expr = \"\"\n",
        "    elif btn == \"⌫\":\n",
        "        state.expr = state.expr[:-1]\n",
        "    elif btn == \"=\":\n",
        "        state.expr = calculate(state.expr).replace(\",\", \"\")\n",
        "    else:\n",
        "        mapping = {\"×\": \"*\", \"÷\": \"/\", \"−\": \"-\"}\n",
        "        state.expr += mapping.get(btn, btn)\n",
